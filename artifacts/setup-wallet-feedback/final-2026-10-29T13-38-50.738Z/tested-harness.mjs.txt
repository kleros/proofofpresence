const hash = (bytes) => createHash("sha256").update(bytes).digest("hex");
const mime = {
  ".html": "text/html",
  ".js": "application/javascript",
  ".css": "text/css",
  ".svg": "image/svg+xml",
  ".woff2": "font/woff2",
};
async function readAssets(directory, prefix = "", result = new Map()) {
  for (const item of await fs.readdir(directory, { withFileTypes: true })) {
    const name = prefix + item.name;
    if (item.isDirectory())
      await readAssets(path.join(directory, item.name), name + "/", result);
    else if (mime[path.extname(name)])
      result.set(name, await fs.readFile(path.join(directory, item.name)));
  }
  return result;
}
function allowed(origin, url, method, body, names) {
  const u = new URL(url);
  if (u.origin !== origin || u.search) return false;
  if (method === "GET")
    return (
      !body &&
      (["/", "/api/bootstrap", "/api/public-catalog"].includes(u.pathname) ||
        names.has(u.pathname.slice(1)))
    );
  if (method === "DELETE")
    return u.pathname === "/api/session" && body === "{}";
  if (method !== "POST") return false;
  let data;
  try {
    data = JSON.parse(body);
  } catch {
    return false;
  }
  if (u.pathname === "/api/auth/challenge")
    return Object.keys(data).length === 1 && data.wallet === actor;
  return (
    u.pathname === "/api/auth/verify" &&
    Object.keys(data).length === 2 &&
    data.nonce === "synthetic-feedback-nonce" &&
    data.signature === "0x" + "7".repeat(130)
  );
}

    if (browser) await browser.close();
    await new Promise((resolve) => server.close(resolve));
    result.completedAt = new Date().toISOString();
    await fs.writeFile(
      path.join(output, "results.json"),
      JSON.stringify(result, null, 2) + "\n",
    );
    console.log(
      JSON.stringify(
        { status: result.status, checks: result.checks.length, output },
        null,
        2,
      ),
    );
  }
}

else if (args.length === 1 && args[0] === "--run-loopback") await run(false);
else if (!args.length)
  console.log(

  );
else
  throw Error(
    "Only --baseline-repro or --run-loopback is accepted; public origin/provider input is unavailable.",
