// Fixed original-transaction diagnostic. No adoption, signer, replay or verifier exceptions.
import fs from "node:fs/promises";
import path from "node:path";
import { fileURLToPath } from "node:url";
import { createHash, randomUUID } from "node:crypto";

const ROOT = fileURLToPath(new URL("../", import.meta.url));
const FIXED = Object.freeze({
  hash: "0xef60cbf9c65883eb3a1daf6f250bd6cc8328ba655270e897ba17870b1d8a06dc",
  child: "0x4542D9b3De78a59d74cbbfe2166e3ef054E42A69",
  factory: "0xEe049D03a007e150ec8ec46B066823043BAcf14E",
  wallet: "0x2bb03593892724d225071068E59ab1E48aF05b55",
  manager: "0xdb9b1e94b5b69df7e401ddbede43491141047db3",
  blockNumber: 26298864,
  eventId: "event-e27f2cde-57bb-4c99-874f-e896f32df9f1",
  eventKey:
    "0x3d540cd492f187403fc07b4dd532e141cb9cbc6dfc568de16896bf046f0cbed7",
  dataHexTextSha256:
    "463c4fa0e1d19ba3f33a5d45cdbb82a7800a8d78af0960a86a1f38d0efa39cdd",
  savedProofSha256:
    "b5aa2b6c3372c2c099fd5c6707c3e7b8c3614455d16e99e7349b228e001b77cb",
});
const RPC = "https://eth.drpc.org";
const REGISTRY = "registry/ethereum-v9-luma-mainnet.json";
const SAVED =
  "artifacts/mainnet-release/original-ef60cbf9-saved-review-readonly.json";
const EXPECTED_STOP =
  "Event creation calldata/identity differs from reviewed configuration";
const sha = (bytes) => createHash("sha256").update(bytes).digest("hex");
const check = (condition, code) => {
  if (!condition) throw Object.assign(new Error(code), { code });
};
const same = (a, b) =>
  typeof a === "string" &&
  typeof b === "string" &&
  a.toLowerCase() === b.toLowerCase();
const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

async function execute() {
  const { ethers } = await import("ethers"),
    { hash } = await import("../src/utils.mjs"),
    engine = await import("../src/upgrade-verification.mjs"),
    { MAINNET_READ_METHODS } = await import("./mainnet/verify-kleros.mjs"),
    { allowedVerificationMethod } = await import(
      "./mainnet/verification-runtime.mjs"
    );
  const registryBytes = await fs.readFile(path.join(ROOT, REGISTRY)),
    manifest = JSON.parse(registryBytes),
    savedBytes = await fs.readFile(path.join(ROOT, SAVED)),
    saved = JSON.parse(savedBytes),
    sourceNames = [
      "scripts/inspect-original-v9-provenance.mjs",
      "src/upgrade-verification.mjs",
      "src/upgrade-artifacts.mjs",
      "scripts/mainnet/runtime-code.mjs",
    ],
    sourceSha256 = Object.fromEntries(
      await Promise.all(
        sourceNames.map(async (name) => [
          name,
          sha(await fs.readFile(path.join(ROOT, name))),
        ]),
      ),
    ),
    proof = {
      schemaVersion: 1,
      kind: "original-v9-unmodified-strict-provenance-diagnostic",
      startedAt: new Date().toISOString(),
      status: "preparing",
      fullProvenanceVerified: false,
      originalHash: FIXED.hash,
      child: FIXED.child,
      factory: FIXED.factory,
      creationBlock: FIXED.blockNumber,
      sourceSha256,
      registrySha256: sha(registryBytes),
      savedReviewProofSha256: sha(savedBytes),
      requests: { total: 0, methods: {}, httpStatuses: {}, rpcErrorCodes: {} },
      financialActions: 0,
      signatures: 0,
      broadcasts: 0,
      adoption: false,
      stateWrites: 0,
      retries: 0,
      minimumRequestStartGapMs: 100,
      limits: [
        "Candidate exists only in process memory; no registry, event, review, state or journal is written.",
        "Actual provider transactions are unmodified. The normal verifier's direct-envelope gate remains authoritative.",
        "An expected strict-envelope stop is partial diagnostic evidence, not full provenance verification.",
        "Raw calldata, authorizations/signatures, permission contexts, event title and signed bytes are never printed or saved.",
      ],
    };
  let provider,
    closed = false,
    tail = Promise.resolve(),
    lastStart = -Infinity,
    stage = "PUBLIC_REVIEW_BINDING";
  try {
    check(
      manifest.environment === "production" &&
        manifest.chain.chainId === 1 &&
        same(manifest.factoryDeployment.factory, FIXED.factory) &&
        sha(savedBytes) === FIXED.savedProofSha256 &&
        saved.draft.eventId === FIXED.eventId &&
        saved.draft.eventKey === FIXED.eventKey &&
        same(saved.draft.wallet, FIXED.wallet) &&
        same(saved.draft.factory, FIXED.factory) &&
        same(saved.draft.submittedHash, FIXED.hash) &&
        saved.transaction.dataHexTextSha256 === FIXED.dataHexTextSha256 &&
        saved.draft.factoryFingerprint === hash(manifest.factoryDeployment),
      "FIXED_SAVED_REVIEW_BINDING_FAILED",
    );
    const methods = new Set(
        MAINNET_READ_METHODS.filter(allowedVerificationMethod),
      ),
      connection = new ethers.FetchRequest(RPC);
    connection.timeout = 20000;
    connection.setThrottleParams({ maxAttempts: 1 });
    connection.getUrlFunc = (request) => {
      const job = tail
        .catch(() => {})
        .then(async () => {
          check(!closed, "READ_PROVIDER_CLOSED");
          const decoded = JSON.parse(new TextDecoder().decode(request.body)),
            items = Array.isArray(decoded) ? decoded : [decoded];
          check(
            items.length === 1 &&
              methods.has(items[0]?.method) &&
              Array.isArray(items[0].params ?? []),
            "FORBIDDEN_RPC_METHOD",
          );
          const wait = 100 - (performance.now() - lastStart);
          if (wait > 0) await sleep(wait);
          check(!closed, "READ_PROVIDER_CLOSED");
          lastStart = performance.now();
          const method = items[0].method;
          proof.requests.total++;
          proof.requests.methods[method] =
            (proof.requests.methods[method] ?? 0) + 1;
          const response = await fetch(RPC, {
              method: "POST",
              headers: { "content-type": "application/json" },
              body: new TextDecoder().decode(request.body),
              redirect: "error",
              signal: AbortSignal.timeout(20000),
            }),
            bytes = new Uint8Array(await response.arrayBuffer());
          proof.requests.httpStatuses[response.status] =
            (proof.requests.httpStatuses[response.status] ?? 0) + 1;
          try {
            const value = JSON.parse(new TextDecoder().decode(bytes));
            for (const item of Array.isArray(value) ? value : [value])
              if (
                typeof item?.error?.code === "number" &&
                Number.isFinite(item.error.code)
              )
                proof.requests.rpcErrorCodes[item.error.code] =
                  (proof.requests.rpcErrorCodes[item.error.code] ?? 0) + 1;
          } catch {
            /* Diagnostics never retain response text. */
          }
          return {
            statusCode: response.status,
            statusMessage: response.status === 200 ? "OK" : "RPC HTTP failure",
            headers: { "content-type": "application/json" },
            body: bytes,
          };
        });
      tail = job.catch(() => {});
      return job;
    };
    provider = new ethers.JsonRpcProvider(connection, 1, {
      staticNetwork: ethers.Network.from(1),
      cacheTimeout: -1,
      batchMaxCount: 1,
    });
    const rawSend = provider.send.bind(provider);
    provider.send = (method, params = []) => {
      check(
        !closed && methods.has(method) && Array.isArray(params),
        "FORBIDDEN_RPC_METHOD",
      );
      return rawSend(method, params); // Never modify params or returned transaction facts.
    };
    for (const name of ["getSigner", "listAccounts", "broadcastTransaction"])
      provider[name] = async () => {
        throw Object.assign(new Error("FORBIDDEN_RPC_METHOD"), {
          code: "FORBIDDEN_RPC_METHOD",
        });
      };
    stage = "CANONICAL_ORIGINAL_AND_WRAPPER";
    check(BigInt(await provider.send("eth_chainId", [])) === 1n, "WRONG_CHAIN");
    const [tx, receipt, block] = await Promise.all([
      provider.send("eth_getTransactionByHash", [FIXED.hash]),
      provider.send("eth_getTransactionReceipt", [FIXED.hash]),
      provider.send("eth_getBlockByNumber", [
        ethers.toQuantity(FIXED.blockNumber),
        false,
      ]),
    ]);
    check(
      tx &&
        receipt &&
        BigInt(tx.chainId) === 1n &&
        BigInt(tx.type) === 4n &&
        BigInt(tx.value) === 0n &&
        same(tx.hash, FIXED.hash) &&
        same(tx.from, FIXED.wallet) &&
        same(tx.to, FIXED.manager) &&
        Number(BigInt(tx.blockNumber)) === FIXED.blockNumber &&
        Number(BigInt(receipt.blockNumber)) === FIXED.blockNumber &&
        BigInt(receipt.status) === 1n &&
        same(receipt.transactionHash, FIXED.hash) &&
        same(tx.blockHash, block.hash) &&
        same(receipt.blockHash, block.hash) &&
        FIXED.blockNumber > saved.draft.reviewBlock,
      "ORIGINAL_CANONICAL_WRAPPER_MISMATCH",
    );
    const wrapper = new ethers.Interface([
        "function redeemDelegations(bytes[] permissionContexts,bytes32[] modes,bytes[] executionCallDatas)",
      ]),
      input = tx.input,
      decoded = wrapper.decodeFunctionData("redeemDelegations", input);
    check(
      input.slice(0, 10) === "0xcef6d209" &&
        decoded[0].length === 1 &&
        decoded[1].length === 1 &&
        decoded[2].length === 1 &&
        decoded[1][0] === ethers.ZeroHash &&
        same(wrapper.encodeFunctionData("redeemDelegations", decoded), input),
      "WRAPPER_EXECUTION_SHAPE_MISMATCH",
    );
    const packed = ethers.getBytes(decoded[2][0]);
    check(
      packed.length > 52 &&
        same(ethers.hexlify(packed.slice(0, 20)), FIXED.factory) &&
        BigInt(ethers.hexlify(packed.slice(20, 52))) === 0n,
      "WRAPPER_TARGET_OR_VALUE_MISMATCH",
    );
    const innerData = ethers.hexlify(packed.slice(52));
    check(
      sha(innerData) === FIXED.dataHexTextSha256 &&
        sha(ethers.getBytes(innerData)) === saved.transaction.dataBytesSha256 &&
        ethers.getBytes(innerData).length === saved.transaction.dataBytes,
      "EXACT_SAVED_REVIEW_BYTES_MISMATCH",
    );
    const artifacts = engine.loadUpgradeableArtifacts(),
      factoryInterface = new ethers.Interface(artifacts.factory.abi),
      parsed = factoryInterface.decodeFunctionData("createEvent", innerData),
      raw = parsed[1],
      config = {
        depositAmount: String(raw.depositAmount),
        eventName: raw.eventName,
        eventStart: Number(raw.eventStart),
        claimDeadline: Number(raw.claimDeadline),
        arbitration: {
          disputeResolver: raw.arbitration.disputeResolver,
          extraData: raw.arbitration.extraData,
          metaEvidenceTemplate: [...raw.arbitration.metaEvidenceTemplate],
          metaEvidenceSlots: raw.arbitration.metaEvidenceSlots,
          policyURI: raw.arbitration.policyURI,
        },
      };
    check(
      parsed[0] === FIXED.eventKey &&
        FIXED.eventKey === ethers.id(FIXED.eventId) &&
        hash(config) === saved.configSha256 &&
        config.arbitration.policyURI ===
          manifest.factoryDeployment.policy.uri &&
        saved.policySha256 === manifest.factoryDeployment.policy.sha256 &&
        same(
          config.arbitration.disputeResolver,
          manifest.arbitration.resolver,
        ) &&
        config.arbitration.extraData === manifest.arbitration.extraData,
      "ORIGINAL_REVIEW_CONFIGURATION_MISMATCH",
    );
    const configType =
        "tuple(uint256 depositAmount,string eventName,uint256 eventStart,uint256 claimDeadline,tuple(address disputeResolver,bytes extraData,string[] metaEvidenceTemplate,bytes metaEvidenceSlots,string policyURI) arbitration)",
      configurationHash = ethers.keccak256(
        ethers.AbiCoder.defaultAbiCoder().encode(
          ["address", configType],
          [FIXED.wallet, config],
        ),
      ),
      factoryEventId = ethers.keccak256(
        ethers.AbiCoder.defaultAbiCoder().encode(
          ["uint256", "address", "address", "bytes32"],
          [1, FIXED.factory, FIXED.wallet, FIXED.eventKey],
        ),
      ),
      created = receipt.logs.filter(
        (log) =>
          same(log.address, FIXED.factory) &&
          same(
            log.topics?.[0],
            factoryInterface.getEvent("EventCreated").topicHash,
          ),
      );
    check(created.length === 1, "EXACT_FACTORY_EVENT_CREATED_REQUIRED");
    const event = factoryInterface.parseLog(created[0]);
    check(
      event &&
        same(event.args.creator, FIXED.wallet) &&
        same(event.args.eventContract, FIXED.child) &&
        event.args.eventKey === FIXED.eventKey &&
        event.args.version === 9n &&
        event.args.eventId === factoryEventId &&
        event.args.configurationHash === configurationHash &&
        !created[0].removed &&
        same(created[0].blockHash, block.hash) &&
        same(created[0].transactionHash, FIXED.hash),
      "ORIGINAL_FACTORY_LOG_MISMATCH",
    );
    const candidate = {
      ...manifest.factoryDeployment,
      eventId: FIXED.eventId,
      contract: FIXED.child,
      owner: FIXED.wallet,
      creator: FIXED.wallet,
      deploymentTxHash: FIXED.hash,
      deploymentBlock: FIXED.blockNumber,
      eventName: config.eventName,
      depositWei: config.depositAmount,
      eventStart: config.eventStart,
      claimDeadline: config.claimDeadline,
      depositOpensAt: Number(BigInt(block.timestamp)),
      depositDeadline: config.eventStart,
      resolver: config.arbitration.disputeResolver,
      arbitrator: manifest.arbitration.arbitrator,
      arbitratorExtraData: config.arbitration.extraData,
      policyURI: config.arbitration.policyURI,
      codeHash: ethers.keccak256(artifacts.proxy.deployedBytecode),
      templateHash: hash([
        config.arbitration.metaEvidenceTemplate,
        config.arbitration.metaEvidenceSlots,
      ]),
      eventKey: FIXED.eventKey,
      factoryEventId,
      configurationHash,
    };
    check(
      same(
        (
          await provider.send("eth_getBlockByNumber", [
            ethers.toQuantity(FIXED.blockNumber),
            false,
          ])
        ).hash,
        block.hash,
      ),
      "PREPARATION_CREATION_REORG",
    );
    proof.preparation = {
      originalReviewBytesMatch: true,
      configurationMatch: true,
      exactFactoryLogMatch: true,
      creationBlockHash: block.hash,
      configurationHash,
      factoryEventId,
      templateHash: candidate.templateHash,
      candidateSha256: hash(candidate),
      sourceProxyCodeHash: candidate.codeHash,
    };
    stage = "UNMODIFIED_VERIFY_V9_DEPLOYMENT";
    await engine.verifyV9Deployment(
      {
        provider,
        manifest,
        chainId: 1,
        production: true,
        governance: manifest.governance,
      },
      candidate,
      engine.V9_MAINNET_SCOPE,
    );
    proof.status = "unexpected-verifier-return";
    proof.stop = { label: "EXPECTED_DIRECT_ENVELOPE_REJECTION_DID_NOT_OCCUR" };
  } catch (error) {
    const expected =
      stage === "UNMODIFIED_VERIFY_V9_DEPLOYMENT" &&
      error?.message === EXPECTED_STOP &&
      error?.code === "UPGRADE_TRUST_MISMATCH";
    proof.status = expected
      ? "stopped-at-strict-direct-envelope-gate"
      : "verification-stopped";
    proof.stop = {
      stage,
      code:
        typeof error?.code === "string" &&
        /^[A-Z][A-Z0-9_]{0,63}$/.test(error.code)
          ? error.code
          : "READONLY_DIAGNOSTIC_UNAVAILABLE",
      label: expected
        ? "EVENT_CREATION_DIRECT_ENVELOPE_REQUIRED"
        : "STOP_BEFORE_EXPECTED_ENVELOPE_OUTCOME",
    };
    if (expected)
      proof.completedBeforeStop = [
        "Registry/profile and approved source/artifact checks",
        "Pinned runtime/code and implementation-slot checks",
        "Pinned implementation family/schema/revision/authority/delay/UUID checks",
        "Complete governance history/roles and pending-operation checks",
        "Initial canonical birth/constructor/initializer checks and historical implementation code",
        "Pinned metadata template reconstruction",
      ];
  } finally {
    closed = true;
    if (provider) {
      try {
        provider.destroy();
      } catch {}
    }
    await tail.catch(() => {});
    proof.notEstablished = [
      "Full child provenance",
      "Normal verifier EventCreated and child-initialization checks after the direct-envelope gate",
      "Current event terms and factory membership checks",
      "Complete initial proxy-upgrade history checks",
      "Final verification-wide birth/governance/head canonical closure",
      "Event adoption or readiness for deposits/manage/financial actions",
    ];
    proof.completedAt = new Date().toISOString();
    const filename = path.join(
      ROOT,
      "artifacts/mainnet-release",
      `original-ef60cbf9-strict-provenance-${new Date().toISOString().replace(/[:.]/g, "-")}-${randomUUID()}.json`,
    );
    await fs.writeFile(filename, JSON.stringify(proof, null, 2) + "\n", {
      flag: "wx",
    });
    console.log(
      JSON.stringify(
        {
          status: proof.status,
          fullProvenanceVerified: false,
          proof: filename,
          requests: proof.requests.total,
          stop: proof.stop,
          signatures: 0,
          broadcasts: 0,
          adoption: false,
        },
        null,
        2,
      ),
    );
    process.exitCode = 1; // This command never reports a full-provenance success.
  }
}

export async function main(argv = process.argv.slice(2)) {
  if (!argv.length)
    return console.log(
      JSON.stringify(
        {
          status: "review-only",
          originalHash: FIXED.hash,
          child: FIXED.child,
          execute:
            "node scripts/inspect-original-v9-provenance.mjs --execute-readonly",
          networkRequests: 0,
          signatures: 0,
          broadcasts: 0,
          adoption: false,
        },
        null,
        2,
      ),
    );
  check(
    argv.length === 1 && argv[0] === "--execute-readonly",
    "EXPLICIT_FIXED_READONLY_FLAG_REQUIRED",
  );
  await execute();
}
if (
  process.argv[1] &&
  path.resolve(process.argv[1]) === fileURLToPath(import.meta.url)
)
  main().catch(() => {
    console.error("FIXED_ORIGINAL_PROVENANCE_PREFLIGHT_FAILED");
    process.exitCode = 1;
  });
